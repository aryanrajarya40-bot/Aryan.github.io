function Crest({ size = 40 }) {
  return (
    <svg width={size} height={size} viewBox="0 0 40 40" style={{ flexShrink: 0 }}>
      <circle cx="20" cy="20" r="19" fill={T.ink} stroke={T.marigold} strokeWidth="1.5" />
      <circle cx="20" cy="20" r="11" fill="none" stroke={T.marigold} strokeWidth="1.1" />
      <circle cx="20" cy="20" r="6" fill={T.peacock} />
      <circle cx="20" cy="20" r="2.2" fill={T.marigold} />
    </svg>
  );
}

function ProgressRing({ percent, size = 84, stroke = 8, from = T.marigold, to = T.peacock, children }) {
  const uid = useId();
  const r = (size - stroke) / 2;
  const c = 2 * Math.PI * r;
  const pct = Math.min(Math.max(percent, 0), 100);
  const offset = c - (pct / 100) * c;
  return (
    <div style={{ position: "relative", width: size, height: size, flexShrink: 0 }}>
      <svg width={size} height={size}>
        <defs>
          <linearGradient id={`g-${uid}`} x1="0%" y1="0%" x2="100%" y2="100%">
            <stop offset="0%" stopColor={from} />
            <stop offset="100%" stopColor={to} />
          </linearGradient>
        </defs>
        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke={T.line} strokeWidth={stroke} />
        <circle
          cx={size / 2} cy={size / 2} r={r} fill="none" stroke={`url(#g-${uid})`}
          strokeWidth={stroke} strokeLinecap="round" strokeDasharray={c} strokeDashoffset={offset}
          transform={`rotate(-90 ${size / 2} ${size / 2})`}
          style={{ transition: "stroke-dashoffset 0.7s ease" }}
        />
      </svg>
      <div style={{ position: "absolute", inset: 0, display: "flex", alignItems: "center", justifyContent: "center", flexDirection: "column" }}>
        {children}
      </div>
    </div>
  );
}

  const styles = {
    primary: { backgroundColor: T.peacock, color: "#fff" },
    marigold: { backgroundColor: T.marigold, color: T.ink },
    ghost: { backgroundColor: "transparent", color: T.ink, border: `1px solid ${T.line}` },

  return (
    <button
      type={type}
      onClick={onClick}
      disabled={disabled}

      style={{ ...styles[variant], ...FONTS.body }}

    </button>
  );
}

export default function App() {
  const [tab, setTab] = useState("dashboard");
  const [students, setStudents] = useState(STUDENTS_SEED);
  const [fees, setFees] = useState(FEE_SEED);
  const [attendance, setAttendance] = useState(ATTENDANCE_SEED);
  const [notices, setNotices] = useState(NOTICES_SEED);

  const [addStudentOpen, setAddStudentOpen] = useState(false);
  const [addNoticeOpen, setAddNoticeOpen] = useState(false);
  const [payModal, setPayModal] = useState(null);
  const [studentDetail, setStudentDetail] = useState(null);

  const classList = [...new Set(students.map((s) => s.cls))].sort();

  const feeFor = (studentId) => fees.find((f) => f.studentId === studentId);
  const attendancePct = (studentId) => {
    const recs = attendance.filter((a) => a.studentId === studentId);
    if (!recs.length) return null;
    const present = recs.filter((r) => r.status === "present").length;
    return Math.round((present / recs.length) * 100);
  };

  const totalCollected = fees.filter((f) => f.status === "paid").reduce((a, f) => a + f.amount, 0);
  const totalDue = fees.filter((f) => f.status === "due").reduce((a, f) => a + f.amount, 0);
  const totalExpected = totalCollected + totalDue;
  const collectedPct = totalExpected ? Math.round((totalCollected / totalExpected) * 100) : 0;
  const dueCount = fees.filter((f) => f.status === "due").length;

  const today = todayStr();
  const todaysRecords = attendance.filter((a) => a.date === today);
  const todaysPct = todaysRecords.length
    ? Math.round((todaysRecords.filter((r) => r.status === "present").length / todaysRecords.length) * 100)
    : null;

  function markPaid(feeId, mode) {
    setFees((prev) => prev.map((f) => (f.id === feeId ? { ...f, status: "paid", paidOn: todayStr(), mode } : f)));
    setPayModal(null);
  }

  function addStudent(data) {
    const id = `s-${Date.now()}`;
    setStudents((prev) => [...prev, { id, ...data }]);
    setFees((prev) => [
      ...prev,
      { id: `f-${id}`, studentId: id, term: CURRENT_TERM, amount: Number(data.feeAmount) || 15000, status: "due", dueDate: "2026-08-31", paidOn: null, mode: null },
    ]);
    setAddStudentOpen(false);
  }

  function addNotice(data) {
    setNotices((prev) => [{ id: `n-${Date.now()}`, date: todayStr(), ...data }, ...prev]);
    setAddNoticeOpen(false);
  }

  function upsertAttendance(records) {
    setAttendance((prev) => {
      const filtered = prev.filter((p) => !records.some((r) => r.id === p.id));
      return [...filtered, ...records];
    });
  }

  const NAV = [
    { id: "dashboard", label: "Dashboard", icon: LayoutDashboard },
    { id: "students", label: "Students", icon: Users },
    { id: "attendance", label: "Attendance", icon: CalendarCheck2 },
    { id: "fees", label: "Fees", icon: IndianRupee },
    { id: "notices", label: "Notices", icon: Megaphone },
  ];

  return (
    <div className="min-h-screen w-full flex" style={{ backgroundColor: T.paper, ...FONTS.body }}>
      <style>{`@import url('https://fonts.googleapis.com/css2?family=Fraunces:opsz,wght@9..144,500;9..144,600;9..144,700&family=Inter:wght@400;500;600;700&family=IBM+Plex+Mono:wght@500;600&display=swap');`}</style>

      {/* desktop sidebar */}
      <aside className="hidden md:flex flex-col w-60 shrink-0 min-h-screen sticky top-0" style={{ backgroundColor: T.ink }}>
        <div className="p-5 flex items-center gap-3" style={{ borderBottom: `1px solid ${T.inkSoft}` }}>
          <Crest size={38} />
          <div>
            <div className="text-sm font-semibold leading-tight" style={{ color: T.paper, ...FONTS.display }}>The Kanha<br />International School</div>
          </div>
        </div>
        <div className="px-5 py-3 text-xs" style={{ color: T.marigold, borderBottom: `1px solid ${T.inkSoft}` }}>
          विद्या ददाति विनयम् <span style={{ color: "#C9C2AE" }}>· knowledge gives humility</span>
        </div>
        <nav className="flex-1 p-3 flex flex-col gap-1">
          {NAV.map((n) => {
            const Icon = n.icon;
            const active = tab === n.id;
            return (
              <button
                key={n.id}
                onClick={() => setTab(n.id)}
                className="flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm font-medium text-left"
                style={{
                  backgroundColor: active ? T.peacockDark : "transparent",
                  color: active ? "#fff" : "#C9C2AE",
                }}
              >
                <Icon size={18} />
                {n.label}
              </button>
            );
          })}
        </nav>
        <div className="p-4 text-xs" style={{ color: "#8891A8", borderTop: `1px solid ${T.inkSoft}` }}>
          Academic Year 2026–27<br />Front Office · Admin
        </div>
      </aside>

      {/* main */}
      <main className="flex-1 min-h-screen pb-24 md:pb-8">
        <header className="md:hidden flex items-center gap-3 p-4" style={{ backgroundColor: T.ink }}>
          <Crest size={32} />
          <div className="text-sm font-semibold" style={{ color: T.paper, ...FONTS.display }}>The Kanha International School</div>
        </header>

        <div className="p-4 sm:p-6 max-w-5xl mx-auto">

            <Dashboard

              totalCollected={totalCollected} totalDue={totalDue} dueCount={dueCount}
              notices={notices} setTab={setTab}

          )}

          {tab === "students" && (
            <Students
              students={students} classList={classList} feeFor={feeFor} attendancePct={attendancePct}
              onAdd={() => setAddStudentOpen(true)} onOpen={setStudentDetail}

          )}

            <Attendance
              students={students} classList={classList} attendance={attendance}
              onSave={upsertAttendance} todaysPct={todaysPct}
            />
          )}

          {tab === "fees" && (
            <Fees
              students={students} fees={fees} collectedPct={collectedPct}
              totalCollected={totalCollected} totalDue={totalDue} onPay={setPayModal}
            />
          )}

          {tab === "notices" && <Notices notices={notices} onAdd={() => setAddNoticeOpen(true)} />}
        </div>
      </main>

      {/* mobile bottom nav */}
      <nav className="md:hidden fixed bottom-0 inset-x-0 z-40 flex justify-around py-2" style={{ backgroundColor: T.ink, borderTop: `1px solid ${T.inkSoft}` }}>
        {NAV.map((n) => {
          const Icon = n.icon;
          const active = tab === n.id;
          return (
            <button key={n.id} onClick={() => setTab(n.id)} className="flex flex-col items-center gap-1 px-2 py-1">

        fee={studentDetail && feeFor(studentDetail.id)} pct={studentDetail && attendancePct(studentDetail.id)}

    </div>

        ))}
      </div>
    </div>
  );
}

/* -------------------------------- students -------------------------------- */
function Students({ students, classList, feeFor, attendancePct, onAdd, onOpen }) {
  const [q, setQ] = useState("");
  const [cls, setCls] = useState("All");

  const filtered = students.filter((s) => {
    const matchQ = s.name.toLowerCase().includes(q.toLowerCase()) || String(s.roll).includes(q);
    const matchCls = cls === "All" || s.cls === cls;
    return matchQ && matchCls;
  });

  return (
    <div>
      <SectionTitle title="Students" right={<Button onClick={onAdd}><UserPlus size={16} /> Add student</Button>} />

      <div className="flex flex-col sm:flex-row gap-3 mb-4">
        <div className="flex-1 flex items-center gap-2 px-3 py-2 rounded-xl" style={{ backgroundColor: "#fff", border: `1px solid ${T.line}` }}>
          <Search size={16} color={T.textSoft} />
          <input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search by name or roll no." className="w-full text-sm focus:outline-none" style={{ color: T.ink }} />
        </div>
        <select value={cls} onChange={(e) => setCls(e.target.value)} className="px-3 py-2 rounded-xl text-sm focus:outline-none" style={inputStyle}>
          <option>All</option>
          {classList.map((c) => <option key={c}>{c}</option>)}
        </select>
      </div>

      <div className="flex flex-col gap-2">
        {filtered.map((s) => {
          const fee = feeFor(s.id);
          const pct = attendancePct(s.id);
          return (
            <Card key={s.id} className="p-3 sm:p-4 flex items-center justify-between gap-3 cursor-pointer" onClick={() => onOpen(s)}>
              <div className="flex items-center gap-3 min-w-0">
                <div className="w-10 h-10 rounded-full flex items-center justify-center shrink-0 text-sm font-bold" style={{ backgroundColor: T.paperAlt, color: T.peacockDark, ...FONTS.mono }}>
                  {s.name.split(" ").map((p) => p[0]).slice(0, 2).join("")}
                </div>
                <div className="min-w-0">
                  <div className="font-semibold text-sm truncate" style={{ color: T.ink }}>{s.name}</div>
                  <div className="text-xs" style={{ color: T.textSoft, ...FONTS.mono }}>{s.cls} · Roll {s.roll}</div>
                </div>
              </div>
              <div className="flex items-center gap-3 shrink-0">
                {pct !== null && <span className="text-xs hidden sm:inline" style={{ color: T.textSoft, ...FONTS.mono }}>{pct}% present</span>}
                <Badge tone={fee?.status === "paid" ? "sage" : "maroon"}>{fee?.status === "paid" ? "Paid" : "Due"}</Badge>
                <ChevronRight size={16} color={T.textSoft} />
              </div>
            </Card>
          );
        })}
        {filtered.length === 0 && <div className="text-sm p-6 text-center" style={{ color: T.textSoft }}>No students match your search.</div>}
      </div>
    </div>
  );
}

/* ------------------------------- attendance ------------------------------- */
function Attendance({ students, classList, attendance, onSave, todaysPct }) {
  const [cls, setCls] = useState(classList[0] || "");
  const [date, setDate] = useState(todayStr());
  const [draft, setDraft] = useState({});
  const [saved, setSaved] = useState(false);

  const roster = students.filter((s) => s.cls === cls);

  useEffect(() => {
    const next = {};
    roster.forEach((s) => {
      const existing = attendance.find((a) => a.studentId === s.id && a.date === date);
      next[s.id] = existing ? existing.status : "present";
    });
    setDraft(next);
    setSaved(false);
    // eslint-disable-next-line
  }, [cls, date]);

  const summary = roster.reduce(
    (acc, s) => { acc[draft[s.id]] = (acc[draft[s.id]] || 0) + 1; return acc; },
    {}
  );

  function save() {
    const records = roster.map((s) => ({ id: `${date}__${s.id}`, studentId: s.id, date, status: draft[s.id] }));
    onSave(records);
    setSaved(true);
  }

  const STATUS_OPTS = [
    { id: "present", label: "Present", tone: "sage", icon: Check },
    { id: "absent", label: "Absent", tone: "maroon", icon: XCircle },
    { id: "leave", label: "Leave", tone: "marigold", icon: Clock3 },
  ];

  return (
    <div>
      <SectionTitle eyebrow={todaysPct === null ? "Today not marked yet" : `Today: ${todaysPct}% present`} title="Mark attendance" />

      <div className="flex flex-col sm:flex-row gap-3 mb-5">
        <select value={cls} onChange={(e) => setCls(e.target.value)} className="px-3 py-2 rounded-xl text-sm focus:outline-none" style={inputStyle}>
          {classList.map((c) => <option key={c}>{c}</option>)}
        </select>
        <div className="flex items-center gap-2 px-3 py-2 rounded-xl" style={inputStyle}>
          <CalendarDays size={16} color={T.textSoft} />
          <input type="date" value={date} onChange={(e) => setDate(e.target.value)} className="text-sm focus:outline-none" style={{ color: T.ink, background: "transparent" }} />
        </div>
      </div>

      <div className="flex flex-col gap-2 mb-5">
        {roster.map((s) => (
          <Card key={s.id} className="p-3 flex items-center justify-between gap-3">
            <div>
              <div className="font-semibold text-sm" style={{ color: T.ink }}>{s.name}</div>
              <div className="text-xs" style={{ color: T.textSoft, ...FONTS.mono }}>Roll {s.roll}</div>
            </div>
            <div className="flex gap-1">
              {STATUS_OPTS.map((opt) => {
                const active = draft[s.id] === opt.id;
                const c = TONES[opt.tone];
                const Icon = opt.icon;
                return (
                  <button
                    key={opt.id}
                    onClick={() => setDraft({ ...draft, [s.id]: opt.id })}
                    className="px-2.5 py-1.5 rounded-lg text-xs font-semibold flex items-center gap-1"
                    style={{ backgroundColor: active ? c.bg : "transparent", color: active ? c.fg : T.textSoft, border: `1px solid ${active ? c.fg : T.line}` }}
                  >
                    <Icon size={13} /> <span className="hidden sm:inline">{opt.label}</span>
                  </button>
                );
              })}
            </div>
          </Card>
        ))}
        {roster.length === 0 && <div className="text-sm p-6 text-center" style={{ color: T.textSoft }}>No students in this class.</div>}
      </div>

      {roster.length > 0 && (
        <div className="flex items-center justify-between flex-wrap gap-3">
          <div className="text-xs" style={{ color: T.textSoft, ...FONTS.mono }}>
            {summary.present || 0} present · {summary.absent || 0} absent · {summary.leave || 0} leave
          </div>
          <Button variant="marigold" onClick={save}>
            {saved ? <><Check size={16} /> Saved</> : "Save attendance"}
          </Button>
        </div>
      )}
    </div>
  );
}

/* ---------------------------------- fees ---------------------------------- */
function Fees({ students, fees, collectedPct, totalCollected, totalDue, onPay }) {
  const [filter, setFilter] = useState("all");
  const nameFor = (id) => students.find((s) => s.id === id);
  const list = fees.filter((f) => filter === "all" || f.status === filter);

  return (
    <div>
      <SectionTitle eyebrow={CURRENT_TERM} title="Fee collection" />

      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 mb-5">
        <Card className="p-4 flex items-center gap-4">
          <ProgressRing percent={collectedPct} size={70}>
            <span className="text-sm font-bold" style={{ color: T.ink, ...FONTS.mono }}>{collectedPct}%</span>
          </ProgressRing>
          <div>
            <div className="text-xs font-semibold uppercase tracking-wide" style={{ color: T.textSoft }}>Collected</div>
            <div className="text-lg font-bold" style={{ color: T.sageDark, ...FONTS.mono }}>{INR(totalCollected)}</div>
          </div>
        </Card>
        <Card className="p-4 flex items-center gap-4">
          <ProgressRing percent={100 - collectedPct} size={70} from={T.maroon} to={T.marigold}>
            <IndianRupee size={20} color={T.maroonDark} />
          </ProgressRing>
          <div>
            <div className="text-xs font-semibold uppercase tracking-wide" style={{ color: T.textSoft }}>Outstanding</div>
            <div className="text-lg font-bold" style={{ color: T.maroonDark, ...FONTS.mono }}>{INR(totalDue)}</div>
          </div>
        </Card>
      </div>

      <div className="flex gap-2 mb-4">
        {["all", "due", "paid"].map((f) => (
          <button
            key={f} onClick={() => setFilter(f)}
            className="px-3 py-1.5 rounded-full text-xs font-semibold capitalize"
            style={{ backgroundColor: filter === f ? T.ink : "#fff", color: filter === f ? "#fff" : T.textSoft, border: `1px solid ${T.line}` }}
          >
            {f}
          </button>
        ))}
      </div>

      <div className="flex flex-col gap-2">
        {list.map((f) => {
          const s = nameFor(f.studentId);
          if (!s) return null;
          return (
            <Card key={f.id} className="p-3 sm:p-4 flex items-center justify-between gap-3">
              <div className="min-w-0">
                <div className="font-semibold text-sm truncate" style={{ color: T.ink }}>{s.name}</div>
                <div className="text-xs" style={{ color: T.textSoft, ...FONTS.mono }}>{s.cls} · {f.term}</div>
              </div>
              <div className="flex items-center gap-3 shrink-0">
                <span className="text-sm font-bold" style={{ color: T.ink, ...FONTS.mono }}>{INR(f.amount)}</span>
                {f.status === "paid" ? (
                  <Badge tone="sage">Paid {fmtDate(f.paidOn)}</Badge>
                ) : (
                  <Button variant="marigold" onClick={() => onPay(f)}>Record payment</Button>
                )}
              </div>
            </Card>
          );
        })}
        {list.length === 0 && <div className="text-sm p-6 text-center" style={{ color: T.textSoft }}>Nothing here.</div>}
      </div>
    </div>
  );
}
